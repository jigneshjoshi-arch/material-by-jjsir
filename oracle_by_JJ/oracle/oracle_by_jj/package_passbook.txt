create or replace package passbook1 as
procedure cpass(act_number number,amt number,c char);
function credit(act_number number,amt number) return number;
function debit(act_number number,amt number) return number;
end passbook1;
/
 create or replace package body passbook1 as
 procedure cpass(act_number number,amt number,c char) is
    a number(10);
 begin
    if c='credit' then
            a := credit(act_number,amt);
            dbms_output.put_line('amount credited');
	elsif c='debit' then
		a := debit(act_number,amt);
		dbms_output.put_line('amount debited');
    end if;
 end cpass;
 function credit(act_number number,amt number) return number is
 begin
    update passbook
    set balance=balance+amt
    where act_number=act_number;

	update passbook
    set credit=amt
    where act_number=act_number;
 return 0;
 end credit;
 function debit(act_number number,amt number) return number is
 begin
    update passbook
    set balance = balance-amt
    where act_number=act_number;

update passbook
    set debit=amt
    where act_number=act_number;

 return 0;
 end debit;
 end passbook1;
/
