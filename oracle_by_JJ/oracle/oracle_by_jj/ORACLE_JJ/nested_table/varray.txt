DECLARE
   type namesarray IS VARRAY(5) OF VARCHAR2(10);
   type grades IS VARRAY(5) OF number(10);
   names namesarray;
   marks grades;
   total number(10);
BEGIN
   names := namesarray('jay', 'vijay', 'rahul', 'virat', 'rohit');
   marks:= grades(98, 97, 78, 87, 92);
   total := names.count;
   dbms_output.put_line('Total '|| total || ' Students');
   FOR i in 1 .. total 
   LOOP
      dbms_output.put_line('Student: ' || names(i) || ' Marks: ' || marks(i));
   END LOOP;
END;
/
