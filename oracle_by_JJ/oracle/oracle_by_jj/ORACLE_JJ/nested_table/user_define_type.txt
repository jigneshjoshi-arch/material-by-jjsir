CREATE OR REPLACE TYPE Address AS OBJECT (
    street     VARCHAR2(50),
    city       VARCHAR2(30),
    postalcode VARCHAR2(10)
);
CREATE TABLE Person (
    person_id   NUMBER(5),
    first_name  VARCHAR2(30),
    last_name   VARCHAR2(30),
    home_address Address  -- using the object type as a column
);
