declare
	jname jstud.jnam%type;

	jgender jstud.gender%type;

	jstream jstud.stream%type;
begin
	jname := '&jname';

	jgender := '&jgender';

	jstream := '&jstream';

	insert into jstud values
	(jname,jgender,jstream);
	
	if jgender = 'M' or jgender = 'm' then
		insert into boys values
		(jname,jstream);
	elsif jgender = 'F' or jgender = 'f' then
		insert into girls values
		(jname,jstream);
	end if;
end;
