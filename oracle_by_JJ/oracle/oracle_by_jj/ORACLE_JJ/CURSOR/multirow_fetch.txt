DECLARE
	J_NAM EMP.ENAME%TYPE;

	jdept emp.deptno%type;

	J_SAL EMP.SAL%TYPE;

	TOT_ROWS NUMBER;

	CURSOR J_EMP IS SELECT ENAME,SAL,deptno FROM EMP where deptno=jdept;
BEGIN

jdept := &jdept;

	OPEN J_EMP;

	LOOP

	FETCH J_EMP INTO J_NAM,J_SAL,jdept;

		EXIT WHEN J_EMP%NOTFOUND;

		DBMS_OUTPUT.PUT_LINE('SALARY OF '||J_NAM||' IS '||J_SAL);
	END LOOP;

	TOT_ROWS := J_EMP%ROWCOUNT;

	DBMS_OUTPUT.PUT_LINE('				');

	DBMS_OUTPUT.PUT_LINE('TOTAL NUMBER OF ROWS ARE  '||TOT_ROWS);

	CLOSE J_EMP;
END;
/
