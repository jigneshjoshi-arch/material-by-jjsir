declare 
	jname jstud.sname%type;
	jgender jstud.gender%type;
	jstream jstud.stream%type;
	
	cursor j1 is select sname,gender,stream from jstud;
begin
	open j1;
	   loop
		fetch j1 into jname,jgender,jstream;
		exit when j1%notfound;
	
		if jgender = 'm' or jgender = 'M' then
			insert into boys values
			(jname,jstream);
		elsif jgender = 'f' or jgender = 'F' then
			insert into girls values
			(jname,jstream);
		end if;
	   end loop;
	close j1;
end;
