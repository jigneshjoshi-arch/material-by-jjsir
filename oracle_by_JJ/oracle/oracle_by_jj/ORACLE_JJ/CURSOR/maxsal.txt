DECLARE
    jsal number(20);
	jmax varchar2(20);
	v varchar2(20);
    CURSOR cur_emp_name IS
      SELECT max(sal) FROM emp group by &v;
BEGIN
    OPEN cur_emp_name;
    LOOP
        FETCH cur_emp_name INTO jsal;
        exit WHEN cur_emp_name%NOTFOUND;
        dbms_output.Put_line('  Salary: '|| jsal);
    END LOOP;
    CLOSE cur_emp_name;
END; 
/
