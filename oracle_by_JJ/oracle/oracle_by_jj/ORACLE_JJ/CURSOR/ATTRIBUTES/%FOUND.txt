DECLARE
	J_NAM EMP.ENAME%TYPE;
	J_SAL EMP.SAL%TYPE;

	CURSOR J_EMP IS SELECT ENAME,SAL FROM EMP;
BEGIN
	OPEN J_EMP;
	
	FETCH J_EMP INTO J_NAM,J_SAL;
	
	WHILE J_EMP%FOUND
	
	LOOP
	
	FETCH J_EMP INTO J_NAM,J_SAL;

	EXIT WHEN J_EMP%NOTFOUND;
	
	DBMS_OUTPUT.PUT_LINE('SALARY OF '||J_NAM||' IS '||J_SAL);
	
	END LOOP;
		
	CLOSE J_EMP;
END;
/
