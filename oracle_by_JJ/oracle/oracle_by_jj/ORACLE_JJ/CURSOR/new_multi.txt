DECLARE
    v_emp_rec emp%ROWTYPE;
	jdept emp.deptno%type;
    CURSOR cur_emp_name IS
      SELECT *
      FROM   emp where deptno=&jdept;
BEGIN
--jdept := &jdept;
    OPEN cur_emp_name;
    LOOP
        FETCH cur_emp_name INTO v_emp_rec;
        exit WHEN cur_emp_name%NOTFOUND;
        dbms_output.Put_line('Name: '
                             || v_emp_rec.ename
                             || '  ::   Salary: '
                             || v_emp_rec.sal);
    END LOOP;
    CLOSE cur_emp_name;
END; 
/
