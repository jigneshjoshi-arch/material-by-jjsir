CREATE OR REPLACE TRIGGER JMSG

	BEFORE INSERT OR DELETE OR UPDATE

	ON jstud
	
	FOR EACH ROW
BEGIN
	IF INSERTING THEN

	DBMS_OUTPUT.PUT_LINE(chr(13));

	DBMS_OUTPUT.PUT_LINE( '1 RECORD INSERTED' );
		

	ELSIF DELETING THEN

	DBMS_OUTPUT.PUT_LINE( 'RECORD DELETED');

	ELSIF UPDATING THEN

	DBMS_OUTPUT.PUT_LINE( 'RECORD UPDATED');

	END IF;
END;
