CREATE OR REPLACE TRIGGER JCHEK

	AFTER UPDATE

	ON EMP 

	FOR EACH ROW

DECLARE

	UPDATE_NOT_ALLOWED EXCEPTION;
	
BEGIN
	IF UPDATING AND :NEW.SAL < :OLD.SAL THEN

            RAISE UPDATE_NOT_ALLOWED;

	END IF;
	
  EXCEPTION

     WHEN UPDATE_NOT_ALLOWED THEN

      RAISE_APPLICATION_ERROR(-20001,'UPDATE FAIL..NEW SAL MUST BE GREATER THEN OLD SAL');

 END;
