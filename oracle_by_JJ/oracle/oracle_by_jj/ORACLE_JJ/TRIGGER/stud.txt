create table std_mark
(sno number(4),
sname char(20),
s1 number(3),
s2 number(3),
s3 number(3),
s4 number(3),
stotal number(5,0),
sper number(5,2))
/
CREATE OR REPLACE TRIGGER std_res
BEFORE INSERT ON std_mark
FOR EACH ROW
BEGIN
    -- Calculate total
    :NEW.stotal := :NEW.s1 + :NEW.s2 + :NEW.s3 + :NEW.s4;

    -- Calculate percentage (assuming the maximum marks per subject is 100)
    :NEW.sper := :NEW.stotal / 4;
END;
/
