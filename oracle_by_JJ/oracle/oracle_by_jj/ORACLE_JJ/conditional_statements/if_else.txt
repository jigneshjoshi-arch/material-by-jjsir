declare
	-- this is if__elsif__end if condition program

	a number(10);
	b number(10);
	c number(10);
begin
	a := &a;
	b := &b;
	c := &c;
		if a>b and a>c then

			dbms_output.put_line('big number is '||a);

		elsif b>a and b>c then 

			dbms_output.put_line('big number is '||b);

		else

			dbms_output.put_line('big number is '||c);

		end if;
end;
