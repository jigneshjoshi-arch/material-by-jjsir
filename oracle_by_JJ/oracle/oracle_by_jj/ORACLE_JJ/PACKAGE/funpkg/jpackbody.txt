create or replace package body jpack as
	function add(n1 in number, n2 in number) return number is
		jtot number(4);
	begin
		jtot := n1 + n2;
		return jtot;
	end add;

	function nmax(n1 in number, n2 in number) return number is
		jmax number(4);
	begin
		if n1>n2 then
			jmax := n1;
		else
			jmax := n2;
		end if;
			return jmax;
	end nmax;
end jpack;
