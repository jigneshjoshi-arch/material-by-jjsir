create or replace package body jpack is
procedure edept(jdept emp.deptno%type,a number,b number,c number) is

f number;
	jnam emp.ename%type;
	jsal emp.sal%type;
	
	cursor c1 is select ename,sal from emp where deptno=jdept;
	
	function jmax(a number,b number, c number) return number is
	ans number;
	
	begin
		ans := greatest(a,b,c);
		return ans;
	end jmax;
begin
	open c1;
loop
	fetch c1 into jnam,jsal;
	
	exit when c1%notfound;
	
	dbms_output.put_line('name is '||jnam||' sal is '||jsal);
end loop;
close c1;
		f:=jmax(a,b,c);
	dbms_output.put_line(f);
 
end edept;
end jpack;
