DECLARE
   TYPE salary IS TABLE OF NUMBER INDEX BY VARCHAR2(20);
   salary_list salary;
   name   VARCHAR2(20);
BEGIN
   -- adding elements to the table
   salary_list('Raj') := 62000;
   salary_list('Mohit') := 75000;
   salary_list('Namrata') := 100000;
   salary_list('Jay') := 78000;

   -- printing the table
   name := salary_list.FIRST;
   WHILE name IS NOT null LOOP
      dbms_output.put_line
      ('Salary of ' || name || ' is ' || salary_list(name));
      name := salary_list.NEXT(name);
   END LOOP;
END;
/

DECLARE
   TYPE salary IS TABLE OF NUMBER INDEX BY VARCHAR2(20);
   salary_list salary;
   name   VARCHAR2(20);
BEGIN
   -- adding elements to the table
   salary_list('Raj') := 62000;
   salary_list('Mohit') := 75000;
   salary_list('Namrata') := 100000;
   salary_list('Jay') := 78000;

   -- printing the table in descending order
   name := salary_list.LAST; -- Initialize name with the last key
   WHILE name IS NOT null LOOP
      dbms_output.put_line
      ('Salary of ' || name || ' is ' || salary_list(name));
      name := salary_list.PRIOR(name); -- Get the previous key (in descending order)
   END LOOP;
END;
/
