CREATE OR REPLACE PROCEDURE JEMP
	(JEMPNO IN EMP.EMPNO%TYPE, JSAL OUT EMP.SAL%TYPE) IS
	
	KNAM EMP.ENAME%TYPE;
BEGIN
	SELECT ENAME, SAL INTO KNAM,JSAL
	FROM EMP WHERE EMPNO = JEMPNO;
	
	DBMS_OUTPUT.PUT_LINE(KNAM ||' '|| ' EXIST ') ;

EXCEPTION

	WHEN NO_DATA_FOUND THEN
	
	DBMS_OUTPUT.PUT_LINE( ' EMPLOYEE DOES NOT EXIST ');
END;
