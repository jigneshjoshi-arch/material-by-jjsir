CREATE OR REPLACE PROCEDURE FINDEMP
	(JEMPNO IN EMP.EMPNO%TYPE) IS

	KNAM EMP.ENAME%TYPE;
BEGIN
	SELECT ENAME INTO  KNAM
	FROM EMP WHERE EMPNO = JEMPNO;

	DBMS_OUTPUT.PUT_LINE( '				');
	DBMS_OUTPUT.PUT_LINE( '				');

	
	DBMS_OUTPUT.PUT_LINE(KNAM ||' '|| ' EXIST ') ;

EXCEPTION

	WHEN NO_DATA_FOUND THEN

	DBMS_OUTPUT.PUT_LINE( '				');
	DBMS_OUTPUT.PUT_LINE( '				');

	DBMS_OUTPUT.PUT_LINE( ' EMPLOYEE DOES NOT EXIST ');
END;
