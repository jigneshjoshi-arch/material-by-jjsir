declare
	vtotsal number(5);
	vdeptno number(2);
begin
	vdeptno := &vdeptno;
	
	select sum(sal) into
	vtotsal from emp
	where deptno = vdeptno;

dbms_output.put_line('  total sal of 	dept :'||vdeptno||' is '||vtotsal);
exception
	when no_data_found then
	dbms_output.put_line('dept number not found');
end;
/
