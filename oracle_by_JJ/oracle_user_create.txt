-- Log in as SYSDBA
sqlplus / as sysdba

-- Create a new user
CREATE USER my_new_user IDENTIFIED BY my_secure_password;

-- Grant basic roles and full DBA role
GRANT CONNECT TO my_new_user;
GRANT RESOURCE TO my_new_user;
GRANT DBA TO my_new_user;

-- Verify the user and their roles
SELECT * FROM DBA_USERS WHERE USERNAME = 'MY_NEW_USER';
SELECT * FROM DBA_ROLE_PRIVS WHERE GRANTEE = 'MY_NEW_USER';
