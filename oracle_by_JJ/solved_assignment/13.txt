create or replace function jsim1(p in number,r in number,n in number) return number is
	simp number(20);
begin
	simp:=p*r*n/100;
	return simp;
end;
/

declare
	p number(10);
	r number(10);
	n number(10);
	ans number(10);
begin
	p:=&p;
	r:=&r;
	n:=&n;
	ans:=jsim1(p,r,n);
	dbms_output.put_line('simple interest is '||ans);
end;
/
