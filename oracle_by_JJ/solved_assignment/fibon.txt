declare
a number := 0;
b number := 1;
temp number;
n number;
begin
	n := &n;
   dbms_output.put_line('fibonacci series is :');
   dbms_output.put_line(a);
   dbms_output.put_line(b);
   for i in 1..n
   loop
      temp:= a + b; --1+1
      a := b;--1
      b := temp;--1
      dbms_output.put_line(temp);
   end loop;
end;
/
