create or replace trigger jchk
after insert
on stud1
for each row

declare

		insert_not_allow exception;
		
begin
	if inserting and :new.sname like 'S%' then
	
		dbms_output.put_line('you can insert record');
	
	else
		raise insert_not_allow;
		
	end if;
exception
		
		when insert_not_allow then
		
		raise_application_error(-20001,'insert not allowed ');
end;
/

insert into stud1
values
('&sname')
/
